# AUFGABEN (QuestLog.h) -- the quest/task list shown in the menu, on the HUD
# and as a marker in the world.
#
# Every line below is a *view* of state the imported pokeemerald scripts
# already keep: a step is done when its flag/variable condition is true, a
# quest is visible once its `start` condition is true and disappears into the
# "erledigt" list once all of its steps are. Nothing here writes anything, so
# editing this file can never break a save -- it only changes what the game
# tells the player to do next.
#
#   quest <QUEST_ID> [main|side]
#   title <one line>
#   desc  <one line, repeatable -- lines are joined>
#   start <condition>          (optional, default: visible from the start)
#   done  <condition>          (optional, default: every step done)
#   step  <text> | <condition> | <target map> [x] [y]
#
# The target is what the world marker points at: the map the objective is on
# (the marker then sits on the warp leading there), optionally with an exact
# tile on that map. See QuestLog.h for the condition grammar.

# ---------------------------------------------------------------------------

# ---------------------------------------------------------------------------
